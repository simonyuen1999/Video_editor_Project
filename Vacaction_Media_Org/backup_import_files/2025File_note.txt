from datetime import datetime
import pytz

# Input UTC timestamp
utc_str = "2025-11-15T21:32:00Z"

# Parse the UTC time
utc_time = datetime.fromisoformat(utc_str)

# Convert to America/New_York timezone
ny_tz = pytz.timezone("America/New_York")
local_time = utc_time.astimezone(ny_tz)

# Format as desired: 'YYYY-MM-DD hh:mm:ss AM/PM'
formatted = local_time.strftime("%Y-%m-%d %I:%M:%S %p")
print(formatted)
